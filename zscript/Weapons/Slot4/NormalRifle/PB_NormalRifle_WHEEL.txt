Weapon consumes tokens in PB_NormalRifle_Functions.zs :: NR_ApplyWheelSelection()

grants GoWeaponSpecialAbility + the chosen Select_* token (same as PB_M41A).


1) Add specialWheel_PB_NormalRifle() near specialWheel_M41A() in zscript/PbWheel/ev_core_special.zsc
----------------------------------------------------------------------------------------------------

	void specialWheel_PB_NormalRifle()
	{
		wheelSpecials.Clear();
		iconScale = (0.65, 0.65);

		PB_SpecialWheel_Mode fire = new ("PB_SpecialWheel_Mode");
		fire.specialIcon = TexMan.CheckForTexture("graphics/pywheel/Carbine_Burst.png", TexMan.Type_Any);
		if (!fire.specialIcon.isValid())
			fire.specialIcon = TexMan.CheckForTexture("RIFXA0", TexMan.Type_Any);
		fire.specialName = "NR_Select_FireMode";
		fire.specialAlias = "Burst / Full-Auto";
		fire.specialPriority = 1;
		fire.iconScale = iconScale;
		wheelSpecials.Push(fire);

		PB_SpecialWheel_Mode dual = new ("PB_SpecialWheel_Mode");
		dual.specialIcon = TexMan.CheckForTexture("graphics/pywheel/Carbine_Dual.png", TexMan.Type_Any);
		if (!dual.specialIcon.isValid())
			dual.specialIcon = TexMan.CheckForTexture("graphics/pywheel/M41DW0.png", TexMan.Type_Any);
		dual.specialName = "NR_Select_DualWield";
		dual.specialAlias = "Dual Wield";
		dual.specialPriority = 2;
		dual.iconScale = iconScale;
		wheelSpecials.Push(dual);

		PB_SpecialWheel_Mode laser = new ("PB_SpecialWheel_Mode");
		laser.specialIcon = TexMan.CheckForTexture("graphics/WeaponWheel/MetalSniper/LaserOn.png", TexMan.Type_Any);
		if (!laser.specialIcon.isValid())
			laser.specialIcon = TexMan.CheckForTexture("graphics/WeaponWheel/MetalSniper/LaserOff.png", TexMan.Type_Any);
		laser.specialName = "NR_Select_Laser";
		laser.specialAlias = "Laser Sight";
		laser.specialPriority = 3;
		laser.iconScale = iconScale;
		wheelSpecials.Push(laser);

		specialWheel_handleMouse();
	}
